if(WIN32)
    if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
        message(FATAL_ERROR "Windows targets are x64 only")
    endif()
    add_library(modern_asio_wasapi STATIC src/wasapi_shared_renderer.cpp)
    target_link_libraries(modern_asio_wasapi PUBLIC modern_asio_core ole32 avrt uuid)
    target_compile_definitions(modern_asio_wasapi PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
    add_executable(modern_asio_smoke tools/shared_mode_smoke.cpp)
    target_link_libraries(modern_asio_smoke PRIVATE modern_asio_wasapi)
    option(MSA_BUILD_ASIO_DLL "Build experimental COM ASIO DLL using official ASIO SDK" OFF)
    if(MSA_BUILD_ASIO_DLL)
        if(NOT ASIO_SDK_DIR OR NOT EXISTS "${ASIO_SDK_DIR}/common/iasiodrv.h")
            message(FATAL_ERROR "ASIO_SDK_DIR must point to Steinberg ASIO SDK root (common/iasiodrv.h)")
        endif()
        add_library(ModernSharedASIO SHARED src/asio_com_driver.cpp)
        target_include_directories(ModernSharedASIO PRIVATE "${ASIO_SDK_DIR}/common")
        target_link_libraries(ModernSharedASIO PRIVATE modern_asio_wasapi ole32 uuid advapi32 winmm)
        if(MSVC)
            # The COM entrypoints are already declared via STDAPI in Windows SDK headers.
            # Export with a .def file instead of redeclaring with __declspec(dllexport).
            target_link_options(ModernSharedASIO PRIVATE "/DEF:${CMAKE_CURRENT_SOURCE_DIR}/src/ModernSharedASIO.def")
        endif()
        target_compile_definitions(ModernSharedASIO PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
        # Manual-only host smoke: loads DLL by path, creates real IASIO buffers,
        # streams 5 seconds, never installs or registers the driver.
        add_executable(modern_asio_host_smoke tools/asio_host_smoke.cpp)
        target_include_directories(modern_asio_host_smoke PRIVATE "${ASIO_SDK_DIR}/common")
        target_link_libraries(modern_asio_host_smoke PRIVATE ole32)
        add_executable(modern_asio_preflight tools/asio_com_preflight.cpp)
        target_include_directories(modern_asio_preflight PRIVATE "${ASIO_SDK_DIR}/common")
        target_link_libraries(modern_asio_preflight PRIVATE ole32)
        if(BUILD_TESTING)
            add_test(NAME modern_asio_com_preflight COMMAND modern_asio_preflight "$<TARGET_FILE:ModernSharedASIO>")
        endif()
    endif()
endif()
