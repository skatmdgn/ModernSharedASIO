cmake_minimum_required(VERSION 3.21)
project(ModernSharedASIO VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(modern_asio_core INTERFACE)
target_include_directories(modern_asio_core INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include")
include(CTest)
if(BUILD_TESTING)
    add_executable(modern_asio_core_tests tests/core_tests.cpp)
    target_link_libraries(modern_asio_core_tests PRIVATE modern_asio_core)
    add_test(NAME modern_asio_core_tests COMMAND modern_asio_core_tests)
endif()
if(WIN32)
    add_library(modern_asio_wasapi STATIC src/wasapi_shared_renderer.cpp)
    target_link_libraries(modern_asio_wasapi PUBLIC modern_asio_core ole32 avrt uuid)
    target_compile_definitions(modern_asio_wasapi PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
    add_executable(modern_asio_smoke tools/shared_mode_smoke.cpp)
    target_link_libraries(modern_asio_smoke PRIVATE modern_asio_wasapi)
endif()
